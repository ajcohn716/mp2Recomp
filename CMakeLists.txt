cmake_minimum_required(VERSION 3.24)
project(MarioParty2Recompiled LANGUAGES C CXX)
set(CMAKE_C_STANDARD 17)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(SDL2 REQUIRED)
find_package(Threads REQUIRED)
set(RT64_STATIC ON CACHE BOOL "" FORCE)
set(RT64_SDL_WINDOW_VULKAN ON CACHE BOOL "" FORCE)
add_subdirectory(rt64)
add_subdirectory(runtime)
file(GLOB RECOMP_SOURCES CONFIGURE_DEPENDS RecompiledFuncs/funcs_*.c)
add_library(mp2_code STATIC ${RECOMP_SOURCES})
target_include_directories(mp2_code PUBLIC N64Recomp/include RecompiledFuncs runtime/librecomp/include)
target_compile_options(mp2_code PRIVATE -fno-strict-aliasing -Wno-implicit-function-declaration)
add_executable(mp2recomp src/main.cpp src/renderer.cpp src/process.cpp src/overlays.cpp src/platform.cpp src/aspMain.cpp)
target_include_directories(mp2recomp PRIVATE
    N64Recomp/include RecompiledFuncs
    rt64/src rt64/src/contrib rt64/src/contrib/plume rt64/src/contrib/hlslpp/include
    rt64/src/contrib/mupen64plus-core/src/api rt64/src/contrib/imgui)
target_compile_definitions(mp2recomp PRIVATE HLSL_CPU RT64_SDL_WINDOW_VULKAN PLUME_SDL_VULKAN_ENABLED)
target_link_libraries(mp2recomp PRIVATE mp2_code librecomp ultramodern rt64 SDL2::SDL2 Threads::Threads ${CMAKE_DL_LIBS})
target_compile_options(mp2recomp PRIVATE -fno-strict-aliasing)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64")
    target_compile_options(mp2recomp PRIVATE -msse4.1)
endif()
target_link_options(mp2recomp PRIVATE -rdynamic)
target_link_options(mp2recomp PRIVATE -Wl,--wrap=osEPiStartDma_recomp)
